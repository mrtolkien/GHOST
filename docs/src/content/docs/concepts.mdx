import { CardGrid, Card } from "@astrojs/starlight/components";

Quick reference for GHOST's core concepts. Each links to its detailed chapter.

<div className="concepts-grid-3up">
  <CardGrid>
  <Card title="OPERATOR">
    The human user. One per installation. Identified by Discord user ID.
  </Card>
  <Card title="GHOST">
    The AI agent. Identity defined by workspace files. One per installation.
  </Card>
  <Card title="Identity">
    BOOT.md, SOUL.md, OPERATOR.md — files that define your GHOST's personality and your
    preferences.<br />[Learn more →](/ghost/identity/)
  </Card>
  <Card title="Provider">
    LLM backend (OpenRouter, Kimi Code, OpenAI OAuth). Configurable model aliases.
    <br />[Learn more →](/ghost/providers/)
  </Card>
  <Card title="Interface">
    Communication transport. Discord is the primary interface. <br />[Learn more
    →](/ghost/interfaces/)
  </Card>
  <Card title="Session">
    A chat thread between OPERATOR and GHOST. Maps 1:1 to a Discord thread. <br />[Learn more
    →](/chat/sessions/)
  </Card>
  <Card title="Knowledge">
    Notes, references, and diary entries stored in SQLite with graph edges and
    embeddings. <br />[Learn more →](/knowledge/knowledge/)
  </Card>
  <Card title="Reflection">
    Automatic knowledge extraction from conversations and research sessions.
    <br />[Learn more →](/knowledge/reflection/)
  </Card>
  <Card title="Skill">
    Workflow file (agentskills.io format) that teaches your GHOST how to handle specific
    tasks. <br />[Learn more →](/skills-and-tools/overview/)
  </Card>
  <Card title="Agent">
    Autonomous background worker for complex multi-step tasks (research, reflection).
    <br />[Learn more →](/agents/introduction/)
  </Card>
  <Card title="Job">
    Cron-scheduled agent task. <br />[Learn more →](/agents/cron/)
  </Card>
  <Card title="Tool">
    A capability the GHOST can invoke during chat (file I/O, search, web fetch, etc.).
    <br />[Learn more →](/skills-and-tools/overview/)
  </Card>
  </CardGrid>
</div>
